# Configurazione
:local githubUrl "INSERISCI_QUI_IL_TUO_URL_RAW"
:local fileName "whitelist_temp.txt"

# Download del file da GitHub
/tool fetch url=$githubUrl mode=https dst-path=$fileName

:delay 2s;

:if ([:len [/file find name=$fileName]] > 0) do={
    :local fileContent [/file get $fileName contents]
    :local lineEnd 0
    :local lastEnd 0

    # Rimuove le vecchie voci di whitelist (tipo FWD)
    /ip dns static remove [/ip dns static find type=FWD]

    :while ($lineEnd < [:len $fileContent]) do={
        :set lineEnd [:find $fileContent "\n" $lastEnd]
        :if ([:len $lineEnd] = 0) do={ :set lineEnd [:len $fileContent] }
        
        :local line [:pick $fileContent $lastEnd $lineEnd]
        
        # Pulizia caratteri \r
        :local cleanLine ""
        :for i from=0 to=([:len $line] - 1) do={
            :local char [:pick $line $i]
            :if ($char != "\r") do={ :set cleanLine ($cleanLine . $char) }
        }
        
        # --- NUOVA LOGICA: Salta commenti e righe vuote ---
        # Prende il primo carattere della riga
        :local firstChar [:pick $cleanLine 0 1]
        
        :if ([:len $cleanLine] > 0 and $firstChar != "#") do={
            
            # Riconoscimento Regex
            :if ([:find $cleanLine "^"] != -1 or [:find $cleanLine "$"] != -1 or [:find $cleanLine "\\"] != -1 or [:find $cleanLine ".*"] != -1) do={
                /ip dns static add regexp=$cleanLine type=FWD forward-to=8.8.8.8
                :put ("Aggiunta Regex: " . $cleanLine)
            } else={
                /ip dns static add name=$cleanLine type=FWD forward-to=8.8.8.8
                :put ("Aggiunto Dominio: " . $cleanLine)
            }
        }
        
        :set lastEnd ($lineEnd + 1)
    }

    /file remove $fileName
    :log info "Whitelist DNS aggiornata (commenti ignorati)."
} else={
    :log error "Errore: impossibile scaricare il file da GitHub."
}
