# ==========================================
# Whitelist DNS - versione semplice compatibile
# ==========================================

:local githubUrl "https://raw.githubusercontent.com/agofa/MikrotikROS_UpdateWhiteList/refs/heads/main/whitelist.txt"
:local fileName "whitelist_temp.txt"

# Scarica il file
/tool fetch url=$githubUrl mode=https dst-path=$fileName

# attende max 10s che il file esista
:local counter 0
:while ([:len [/file find name=$fileName]] = 0 and $counter < 10) do={
    :delay 1s
    :set counter ($counter + 1)
}

# Se il file esiste...
:if ([:len [/file find name=$fileName]] > 0) do={

    :local data [/file get $fileName contents]
    :local lastEnd 0
    :local lineEnd 0


# Cancella tutte le whitelist precedenti (compatibile ROS7)
:local ids [/ip dns static find where type=fwd]

:if ([:len $ids] > 0) do={
    /ip dns static remove $ids
}


    # Scorre il file riga per riga
    :while ($lastEnd < [:len $data]) do={

        :set lineEnd [:find $data "\n" $lastEnd]
        :if ($lineEnd = -1) do={ :set lineEnd [:len $data] }

        :local line [:pick $data $lastEnd $lineEnd]

        # --- RIMUOVE \r SE PRESENTE ---
        :if ([:len $line] > 0) do={
            :local lastChar [:pick $line ([:len $line] - 1)]
            :if ($lastChar = "\r") do={
                :set line [:pick $line 0 ([:len $line] - 1)]
            }
        }

        # Se non è vuota, aggiungi DNS
        :if ([:len $line] > 0) do={
            /ip dns static add name=$line type=FWD forward-to=8.8.8.8
        }

        :set lastEnd ($lineEnd + 1)
    }

    /file remove $fileName
    :log info "[Whitelist] Aggiornata (versione semplice compatibile)"

} else={
    :log error "[Whitelist] Impossibile scaricare file!"
}
