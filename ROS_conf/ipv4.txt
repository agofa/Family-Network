# eliminare regole esistenti
/ip firewall filter remove [find dynamic=no]
/ip firewall raw remove [find dynamic=no]

# -----------------------------------------------------------------------------
# 1. TABELLA RAW: Il "Buttafuori" (Protezione precoce a bassa CPU)
# -----------------------------------------------------------------------------
/ip firewall raw

# Drop pacchetti TCP "impossibili" (senza flag)
add chain=prerouting action=drop in-interface=ether1 protocol=tcp tcp-flags=!fin,!syn,!rst,!psh,!ack,!urg comment="RAW: Drop TCP malformati"

# Anti-Spoofing: Blocca chi bussa alla WAN fingendo di avere IP della tua LAN
add chain=prerouting action=drop in-interface=ether1 src-address=192.168.0.0/16 comment="RAW: Drop Spoofing 192.168"
add chain=prerouting action=drop in-interface=ether1 src-address=10.0.0.0/8 comment="RAW: Drop Spoofing 10.x"
add chain=prerouting action=drop in-interface=ether1 src-address=172.16.0.0/12 comment="RAW: Drop Spoofing 172.16"

# RP-Filter (Reverse Path Filter)
/ip settings set rp-filter=loose

# -----------------------------------------------------------------------------
# 2. TABELLA FILTER: La "Logica" (Gestione connessioni e sicurezza)
# -----------------------------------------------------------------------------
/ip firewall filter

# --- CATENA INPUT: Protezione del Router stesso ---
add chain=input action=accept connection-state=established,related,untracked comment="INPUT: Accetta stabiliti/untracked"
add chain=input action=drop connection-state=invalid comment="INPUT: Drop invalidi"
add chain=input action=accept protocol=icmp limit=5,10:packet comment="INPUT: ICMP (Ping) limitato"
add chain=input action=accept in-interface=bridge comment="INPUT: LAN -> Router"
# add chain=input action=accept protocol=tcp dst-port=2222,18291 in-interface=ether1 connection-state=new comment="INPUT: Gestione remota WAN"
# add chain=input action=accept protocol=tcp dst-port=2222,18291 src-address=192.168.10.0/24 comment="INPUT: Gestione LAN"
add chain=input action=drop in-interface=ether1 log=yes log-prefix="WAN-DROP-MGMT" comment="INPUT: DROP FINALE WAN"


# --- CATENA FORWARD: Traffico che attraversa il router verso i tuoi dispositivi ---
add chain=forward action=fasttrack-connection connection-state=established,related comment="FORWARD: Fasttrack (Ottimizzazione CPU)"
add chain=forward action=accept connection-state=established,related,untracked comment="FORWARD: Accetta stabiliti"
add chain=forward action=drop connection-state=invalid comment="FORWARD: Drop invalidi"

# Protezione LAN: Blocca connessioni NUOVE da WAN che non sono port forwarding (DST-NAT)
add chain=forward action=drop connection-nat-state=!dstnat connection-state=new in-interface=ether1 comment="FORWARD: Drop WAN non nattata"

# Permessi in uscita e interni
add chain=forward action=accept in-interface=bridge out-interface=ether1 comment="FORWARD: LAN -> WAN (Navigazione)"
add chain=forward action=accept in-interface=bridge out-interface=bridge comment="FORWARD: Traffico interno tra dispositivi LAN"

# Policy di chiusura: Se un pacchetto non è tra quelli sopra, lo uccidiamo qui
add chain=forward action=drop in-interface=ether1 log=yes log-prefix="WAN-DROP-LAN" comment="FORWARD: DROP FINALE (Sicurezza totale WAN->LAN)"
